## Creamos una función para detectar si tiene propiedades ZM
tiene_zm <- function(atrib) {
  coords <- atrib[7]
  # Detectar si hay columnas Z o M
  return(grepl("3D", coords))
}

# función de validación de dominios segun la tabla, considerando los NA y "" como ERROR
valida_dominio <- function(valores, dominio) {
  es_na_o_vacio <- is.na(valores) | valores == ""
  fuera_dom <- !(valores %in% dominio)
  invalido <- es_na_o_vacio | fuera_dom
  list(ok_global = !any(invalido),
       idx_invalidos = which(invalido)
  )
}

#### CILCO DE VALIDACION SEGUN PLANILLA ####
for (shp in shapes_filtrados) {
  # shp <- shapes_filtrados[2] # MANTENER DESACTIVADO, USARLO SOLO PARA VER COMO VA ANALISISANDO EL CICLO
  nombre_shape <- tools::file_path_sans_ext(basename(shp))
  prefijo_usado <- prefijo_por_shape[[nombre_shape]]
  plantilla <- plantillas[[prefijo_usado]]
  cat(bold(black( "Revisando shape:", nombre_shape, "\n")))
  datos <- st_read(shp, quiet = TRUE) # Generando el shape a validar
  
  # Revisar Coordendas
  crs_shape <- sf::st_crs(datos) # rescatamos el sistema de coordenadas que tiene el shape
  if (!is.na(crs_shape$epsg) && crs_shape$epsg == 32719) {
    cat(bold(green(paste0("Coordenadas CORRECTAS (WGS84 / UTM 19S, EPSG:32719) en '", nombre_shape, "'.\n"))))
  } else {
    cat(bold(red(paste0("Coordenadas INCORRECTAS en '", nombre_shape, "'. Tiene: ",
                        ifelse(is.na(crs_shape$epsg), "NA", crs_shape$epsg), "\n"))))
  }
  
  atributos <- st_drop_geometry(datos)  # convertimos nuestro shape a una tabla, al eliminar la geometria
  campos <- data.frame(nombre = names(atributos), tipo = sapply(atributos, function(x) class(x)[1]), stringsAsFactors = FALSE)
  
  # Llamar a ogrinfo para obtener estructura, por lo que entendí, es un ejecutable de GDAL (QGIS)
  cmd <- paste0("\"", ogrinfo_path, "\" -so -al \"", shp, "\"")
  salida <- system(cmd, intern = TRUE)
  
  if (tiene_zm(salida)) { # Aplicamos la funcion ZM al shape (datos) para validar
    cat(bold(red(paste0("El shape '", nombre_shape, "' tiene dimensiones Z/M en la geometría. (", salida[7], ")\n"))))
  } else {
    cat(bold(green(paste0("El shape '", nombre_shape, "' es 2D (sin dimensiones Z/M). (", salida[7], ")\n"))))
  }
  
  campos_lineas <- salida[grepl("^\\s*\\w+\\s*:\\s*\\w+\\s*\\(\\d+(\\.\\d+)?\\)$", salida)]
  
  campos_info <- stringr::str_match(campos_lineas, "^\\s*(\\S+):\\s+(\\S+) \\((\\d+)\\.?((\\d+)?)\\)")
  
  info_df <- data.frame(
    nombre = campos_info[,2],
    tipo_ogr = campos_info[,3],
    ancho = as.integer(campos_info[,4]),
    decimales = as.integer(ifelse(campos_info[,6] == "", 0, campos_info[,6])),
    stringsAsFactors = FALSE
  )
  
  # Ajustar a los tipos a los indicados en la plantilla para más fácil comparación
  map_tipos <- function(x) {
    if (x %in% c("String", "Character")) return("character")
    if (x %in% c("Integer")) return("integer")
    if (x %in% c("Real", "Float", "Double")) return("numeric")
    return(tolower(x))
  }
  info_df$tipo_esperado <- sapply(info_df$tipo_ogr, map_tipos)  # Aplicamos la función map tipos a cada linea de datos de los tipos leidos con GDAL
  
  resultado <- plantilla %>%
    left_join(campos, by = "nombre") %>%
    left_join(info_df, by = "nombre")
  
  resultado$Existe_Campo <- ifelse(is.na(resultado$tipo.y), "No", "Sí")
  resultado$Coincide_Tipo_Campo <- resultado$tipo.x == resultado$tipo.y
  
  # Validación del largo de los campos
  resultado$Longitud_Definida_Correcta <- NA
  for (i in seq_len(nrow(resultado))) {
    if (!is.na(resultado$longitud[i]) && !is.na(resultado$ancho[i]) && resultado$Existe_Campo[i] == "Sí") {
      resultado$Longitud_Definida_Correcta[i] <- resultado$longitud[i] == resultado$ancho[i]
    }
  }
  
  # Validación de decimales
  resultado$Decimales_Definidos_Correctos <- NA
  for (i in seq_len(nrow(resultado))) {
    if (!is.na(resultado$decimales.x[i]) && !is.na(resultado$decimales.y[i]) && resultado$Existe_Campo[i] == "Sí") {
      resultado$Decimales_Definidos_Correctos[i] <- resultado$decimales.x[i] == resultado$decimales.y[i]
    }
  }
  
  # pegar valores en la tabla
  campos_plantilla <- plantilla$nombre
  campos_shape <- names(atributos)
  coincide_orden <- identical(campos_plantilla, campos_shape)
  resultado$Orden_Correcto <- coincide_orden
  
  ## Ajuste de nombres para entender mejor la tabla
  resultado <- resultado %>%
    rename(
      Tipo_Plant = tipo.x,
      Tipo_Shp = tipo.y,
      Dec_Plant = decimales.x,
      Dec_Shp = decimales.y,
      Coin_Tipo_Campo = Coincide_Tipo_Campo,
      Long_Plant = longitud,
      Dec_Correcto = Decimales_Definidos_Correctos,
      Long_Correcta = Longitud_Definida_Correcta,
      Long_Shp = ancho
    ) %>%
    select( # filtrar solo los campos que queremos, elimando los temporales
      nombre,
      Tipo_Plant,
      Tipo_Shp,
      Existe_Campo,
      Coin_Tipo_Campo,
      Long_Plant,
      Long_Shp,
      Long_Correcta,
      Dec_Plant,
      Dec_Shp,
      Dec_Correcto,
      orden_correcto = Orden_Correcto #everything() agregar esto ultimo si se quiere mantener el restos de los valores de la tabla
    )
  
  print(resultado)
  if (coincide_orden) {
    cat(bold(green(paste0("El shape '", nombre_shape, "' tiene el orden CORRECTO \n"))))
  } else {
    cat(bold(red(paste0("El shape '", nombre_shape, "' tiene el orden INCORRECTO \n"))))
    cat(bold(red((paste0(campos_plantilla, collapse = " / ")),"\n",(paste0(campos_shape, collapse = " / ")), "\n")))
  }
  
  # Validación de datos segun el protocolo conaf
  dominios_ok <- TRUE
  for (campo in names(dominios_conaf)) {
    if (campo %in% names(atributos)) {
      res_dom <- valida_dominio(atributos[[campo]], dominios_conaf[[campo]])
      if (!res_dom$ok_global) {
        dominios_ok <- FALSE
        cat(bold(red("Campo '", campo, "' contiene valores fuera del dominio o vacíos (NA/'')\n")))
      }
    }
  }
  if (dominios_ok) cat(bold(green("Dominios de atributos OK (sin vacíos ni valores diferentes)\n")))
  
  # Verificar Sup_ha vs geometría
  if ("Sup_ha" %in% names(atributos)) {
    area_geom <- round(as.numeric(st_area(datos))/10000,2)
    dif_rel <- abs(area_geom - atributos$Sup_ha)/pmax(area_geom, 1e-6)
    if (any(dif_rel > 0.01, na.rm = TRUE)) {
      cat(bold(red("Hay diferencias entre el campo Sup_ha y el área recalcualada\n")))
    } else {
      cat(bold(green("No hay diferencias entre Sup_ha con el área recalculada\n")))
    }    
  }
  
  # Revisar coordenadas
  if (all(c("Coord_X", "Coord_Y") %in% names(atributos))) {
    # centroide de cada shape que tiene campos de coordenadas (redondeado a enteros)
    cent <- st_centroid(datos)
    coords_cent <- st_coordinates(cent)
    cent_X <- round(coords_cent[, "X"])  # redondear a entero
    cent_Y <- round(coords_cent[, "Y"])  # redondear a entero
    
    # valores de los campos, asegurando que sean numéricos en caso que esten malos en una primera revisión
    campo_X <- as.numeric(atributos$Coord_X)
    campo_Y <- as.numeric(atributos$Coord_Y)
    coinc_X <- cent_X == campo_X
    coinc_Y <- cent_Y == campo_Y
    
    # error si no coinciden
    idx_malos <- which(!coinc_X | !coinc_Y)
    
    if (length(idx_malos) == 0) {
      cat(bold(green("Coord_X / Coord_Y coherentes con las coordenadas recalculadas \n")))
    } else {
      cat(bold(red("Coord_X / Coord_Y diferentes del centroide calculado o campos vacíos: ", length(idx_malos), "\n")))
      cat("     Filas (primeros 10):", paste(head(idx_malos, 10), collapse = ", "), "\n")
    }
  }
  
  # Revisar acentos u otro caracter (agregar donde corresponda)
  campos_texto <- names(atributos)[sapply(atributos, is.character)]
  
  if (length(campos_texto) > 0) {
    formato_txt <- FALSE
    doble_esp <- FALSE
  
    for (campo_txt in campos_texto) {
      valores <- atributos[[campo_txt]]
      
      # listar los caracteres que se quiera buscar
      tiene_txt <- grepl("[áéíóúÁÉÍÓÚ'àèìòùÀÈÌÒÙ´`]", valores)
      idx_invalidos <- which(!is.na(valores) & tiene_txt)
      
      if (length(idx_invalidos) > 0) {
        formato_txt <- TRUE
        cat(bold(red(
          paste0("Campo '", campo_txt,
                 "' contiene caracteres invalidos en las filas: ",
                 paste(head(idx_invalidos, 10), collapse = ", "),
                 if (length(idx_invalidos) > 10) " ..." else "", "\n")
        )))
      }
      # Dobles espacios
      tiene_doble_esp <- grepl("  ", valores)
      idx_doble_esp <- which(!is.na(valores) & tiene_doble_esp)
      
      if (length(idx_doble_esp) > 0) {
        doble_esp <- TRUE
        cat(bold(red(
          paste0("Campo '", campo_txt,
                 "' contiene dobles espacios en las filas: ",
                 paste(head(idx_doble_esp, 10), collapse = ", "),
                 if (length(idx_doble_esp) > 10) " ..." else "", "\n")
        )))
      }
    }
    
    if (!formato_txt) {
      cat(bold(green("No hay caracteres invalidos en los campos de texto del shape \n")))
    }
    if (!doble_esp) {
      cat(bold(green("No hay dobles espacios en campos de texto.\n")))
    }
  } else {
    cat(bold(black("El shape no tiene campos de texto para validar caracteres o dobles espacios.\n")))
  }
  
  geom_valid <- st_is_valid(datos, NA_on_exception = TRUE)
  invalid_geom <- sum(!geom_valid, na.rm = TRUE)
  if (invalid_geom > 0) {
    cat(bold(red(invalid_geom, " geometrías inválidas (self-intersection, etc.)\n\n")))
  } else {
    cat(bold(green("Todas las geometrías son válidas\n\n")))
  }
  resultados_lista[[nombre_shape]] <- resultado
}
